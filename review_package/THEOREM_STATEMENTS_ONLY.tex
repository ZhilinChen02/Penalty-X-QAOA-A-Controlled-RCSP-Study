\noindent This packet contains statements only. It does not claim that an
external review has occurred. Let $\mathcal F$ be uniform among size-$M$
subsets of $[N]$ and $\phi=M/N$, except where an explicit RCSP model is stated.

\paragraph{T1 (fixed phase-sensitive).}
For fixed membership-query phases $\gamma_t$ and feasible-set-independent
nonquery operations,
\[
\mathbb E_{\mathcal F}P_{\mathcal F}\le
\min\left\{1,\left(1+\sum_t|e^{-i\gamma_t}-1|\right)^2\phi\right\}.
\]

\paragraph{T2 (adaptive hard cap).}
A purified adaptive membership-query protocol using at most $q$ queries on
every path satisfies
\[\mathbb E P_{\rm success}\le\min\{1,(2q+1)^2\phi\}.\]

\paragraph{T3 (trained total query).}
T2 applies with the pathwise total of training, sampling, feedback, and final
membership queries; it does not bound final trained depth.

\paragraph{T4 (expected-query truncation).}
If $\bar q=\mathbb E Q$, then for every integer $T\ge0$,
\[
\mathbb E P\le\min\left\{1,\frac{\bar q}{T+1}+
\min\{1,(2T+1)^2\phi\}\right\}.
\]

\paragraph{T5 (raw-density impossibility).}
No universal positive-constant explicit-RCSP lower bound can depend only on
$\phi_{\rm state}^{-1/2}$: an explicit $m$-edge chain has
$\phi_{\rm state}=2^{-m}$ and output/traversal cost $\Theta(m)$.

\paragraph{T6 (serial padding).}
Private $r$-edge subdivision with exact rational attribute splitting preserves
routes, feasibility and optima and satisfies
\[\phi_{\rm state}(I^{(r)})=2^{-(r-1)}\phi_{\rm state}(I).\]
Per-coefficient bit length grows $O(\log r)$; explicit topology grows
$\Theta(r)$.

\paragraph{T7 (explicit attribute-query RCSP).}
For a known $K$-branch topology and a counted random-access explicit status
array with $M$ marked branches, the nontrivial fixed-target query complexity is
$\Theta_\tau(\sqrt{K/M})$. This is not a RAM-time or compact-input lower bound.

\paragraph{T8 (posterior structure).}
For classical pre-search structure $S$, define
\[
\overline\Pi_s=\mathbb E[\Pi_{\mathcal F}\mid S=s],\quad
\lambda_s=\|\overline\Pi_s\|_\infty,\quad
\Lambda=\mathbb E_S\lambda_S.
\]
Then a hard-cap-$q$ residual membership-query protocol satisfies
\[\mathbb E P\le\min\{1,(2q+1)^2\Lambda\}.\]

\paragraph{T9 (finite classical advice).}
If $|\operatorname{supp}S|\le2^b$, then
\[
\Lambda\le\min\{1,2^b\phi\},\qquad
\mathbb E P\le\min\{1,(2q+1)^2 2^b\phi\}.
\]

\paragraph{T10 (one-shot quantum advice).}
An analogous dimension factor holds for one pre-search
$\mathcal F$-dependent state on a total accessible $d$-dimensional register,
including mixed advice and an inaccessible purification. Refreshed and
interactive advice are excluded.

\paragraph{T11 (effective quantities).}
For $0<\phi$, let
\[b_{\rm eff}=\log_2(\Lambda/\phi),\qquad K_{\rm eff}=M/\Lambda.\]
Then $N/K_{\rm eff}=2^{b_{\rm eff}}$, and target average success $\tau$ requires
\[
b_{\rm eff}+2\log_2(2q+1)\ge\log_2(\tau/\phi).
\]
This is necessary, not sufficient or an implementation-time result.
